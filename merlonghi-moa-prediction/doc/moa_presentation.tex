      \begin{itemize}
        \item \textbf{Host:} Laboratory for Innovation Science at Harvard (LISH), with the Broad Institute's Connectivity Map (LINCS);
        \item \textbf{Platform:} Kaggle -- \texttt{\url{https://kaggle.com/c/lish-moa}}; 
        \item \textbf{Timeline:} launched and closed in 2020 (final deadline Nov.\ 30, 2020);
        \item \textbf{Scale:} several thousand participating teams ($\sim$4{,}300 reported by published write-ups);
        \item \textbf{Data origin:} L1000 assay -- a high-throughput assay measuring how human cell lines respond transcriptionally to a chemical or genetic perturbation.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Why it matters}
        Mapping a compound's gene-expression and cell-viability signature to its mechanism of action accelerates \textbf{drug discovery} and \textbf{repurposing}, replacing slow, costly wet-lab characterisation with a computational first pass.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------
% 3. THE PROBLEM
% ---------------------------------------------------------------
\begin{frame}{The Problem}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item \textbf{Task:} multi-label classification of \textbf{206 mechanisms of action (MoA)}
        \item \textbf{Input:} gene-expression (\texttt{g-}) and cell-viability (\texttt{c-}) profiles
        \item \textbf{Metric:} mean multi-label binary log loss, clipped to avoid $\log(0)$
        \item Control rows (\texttt{cp\_type == ctl\_vehicle}) carry \textbf{no MoA} and are excluded from training
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Why an ensemble?}
        Dense nets and tree models fail differently on rare positives $\Rightarrow$ blending several architectures on different preprocessing variants reduces variance.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

def encode_categoricals(df, dummy_cols, reference_columns=None):
    """One-hot encodes cp_time/cp_dose, aligned to the training columns."""

\section{Training}

% ---------------------------------------------------------------
% 15. METRIC & LOSS
% ---------------------------------------------------------------
\begin{frame}{Metric, Loss Variants and Per-fold Training Routine}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Evaluation metric}
      \[
        \mathcal{L} = -\frac{1}{N \cdot 206}\sum y\log \hat y + (1-y)\log(1-\hat y)
      \]
      predictions clipped to $[\epsilon, 1-\epsilon]$, with $\epsilon = 10^{-15}$.
      \\
      \vspace{8mm}
      \textbf{Optional variants}
      \begin{itemize}
        \item \textbf{Focal loss:} down-weights easy examples, useful with rare positives
        \item \textbf{Label smoothing} (0.001): $y(1-\epsilon) + 0.5\epsilon$
        \item Validation loss \textbf{always} uses non-smoothed targets $\to$ honest comparison
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item \textbf{AMP} (\texttt{torch.amp.GradScaler}) for speed and stability
        \item \textbf{Early stopping} (patience = 8 epochs)
        \item \textbf{ReduceLROnPlateau} scheduler
        \item \textbf{In-RAM checkpointing}: best \texttt{state\_dict} cloned to CPU, written to disk once at the end
        \item Explicit \texttt{torch.cuda.empty\_cache()} so sibling workers reclaim VRAM
      \end{itemize}
      \begin{alertblock}{Why it matters}
        Avoids dozens of slow disk writes per fold and prevents GPU memory creep across 45 parallel jobs.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------
% 17. PREPROCESSING
% ---------------------------------------------------------------
\begin{frame}{Two Preprocessing Pipelines}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{PCA variant (MLP, ResNet)}
        VarianceThreshold $\to$ RankGauss (\texttt{QuantileTransformer}) $\to$ PCA(50) on \texttt{g-}, PCA(15) on \texttt{c-} $\to$ concat + raw dummies
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Raw variant (FT-Transformer, CatBoost)}
        VarianceThreshold $\to$ RankGauss, \textbf{no PCA} -- keeps one token/split per original feature
      \end{block}
    \end{column}
  \end{columns}
  \vspace{2mm}
  \small Both fitted \textbf{per fold} on the training split only, cached to disk as \texttt{.joblib}, and reused for CatBoost to avoid refitting.
\end{frame}

\section{Cross-Validation \& Ensembling}

% ---------------------------------------------------------------
% 18. NN CV SCALE
% ---------------------------------------------------------------
\begin{frame}[fragile]{NN Cross-Validation at Scale}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item $3$ models $\times$ $3$ seeds $\times$ $5$ folds $=$ \textbf{45 jobs}
        \item \texttt{joblib} + \texttt{loky} backend, GPUs assigned \textbf{round-robin}
        \item 2 GPUs $\times$ 2 jobs/GPU $=$ \textbf{4 parallel workers}
        \item Predictions averaged over the 3 seeds per (fold, model)
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{lstlisting}
NN_JOBS_PER_GPU = 2
N_JOBS_NN = N_GPUS * NN_JOBS_PER_GPU

Parallel(n_jobs=N_JOBS_NN,
         backend="loky")(
  delayed(_train_nn_job)(...)
  for job in jobs
)
      \end{lstlisting}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------
% 19. BLENDING METHODOLOGY
% ---------------------------------------------------------------
\begin{frame}{Ensemble Blending: Methodology}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Weight search}
      \begin{itemize}
        \item Softmax-reparameterised weights on the simplex
        \item \textbf{Nelder--Mead}, not SLSQP: avoids a zero finite-difference gradient stalling the optimizer at $x_0$
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Nested CV}
      \begin{itemize}
        \item Weights fit on $k-1$ folds, scored on the held-out fold
        \item Prevents an over-optimistic blend score
        \item Final weights = mean across folds
      \end{itemize}
    \end{column}
  \end{columns}
  \begin{exampleblock}{Two scores reported}
    \emph{naive} OOF blend score (same data used to fit the weights) vs.\ \emph{honest} nested-CV score
  \end{exampleblock}
\end{frame}

% 26. KEY DECISIONS
% ---------------------------------------------------------------
\begin{frame}{Key Engineering Decisions}
  \begin{itemize}
    \item \textbf{Per-feature RankGauss}, PCA only where it helps (dense nets), never for trees
    \item \textbf{Multi-seed bagging} smooths single-run noise before blending
    \item \textbf{Round-robin GPU scheduling} keeps multi-GPU training balanced
    \item \textbf{In-RAM checkpoints} minimise I/O across 45 parallel jobs
    \item \textbf{Nested-CV blend weights} avoid leaking validation information into the final score
  \end{itemize}
\end{frame}
