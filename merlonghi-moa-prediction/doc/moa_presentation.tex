\section{The Challenge}

% ---------------------------------------------------------------
% ABOUT THE COMPETITION
% ---------------------------------------------------------------
\begin{frame}{About the Competition}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \begin{itemize}
        \item \textbf{Host:} Laboratory for Innovation Science at Harvard (LISH), with the Broad Institute's Connectivity Map (LINCS);
        \item \textbf{Platform:} Kaggle -- \url{https://kaggle.com/c/lish-moa};
        \item \textbf{Timeline:} launched and closed in 2020 (final deadline Nov.\ 30, 2020);
        \item \textbf{Scale:} 4,373 teams on the final leaderboard;
        \item \textbf{Format:} code competition -- a notebook is submitted and re-run by Kaggle on a hidden test set;
        \item \textbf{Data origin:} gene expression from the L1000 assay, which measures how human cell lines respond transcriptionally to a chemical perturbation; cell viability from the PRISM assay.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Why it matters}
        Mapping a compound's gene-expression and cell-viability signature to its mechanism of action accelerates \textbf{drug discovery} and \textbf{repurposing}, replacing slow, costly wet-lab characterisation with a computational first pass.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------
% THE PROBLEM AND THE DATA
% ---------------------------------------------------------------
\begin{frame}{The Problem and the Data}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item \textbf{Task:} multi-label classification of \textbf{206 mechanisms of action (MoA)}
        \item \textbf{Input:} 772 gene-expression (\texttt{g-}) and 100 cell-viability (\texttt{c-}) features, plus \texttt{cp\_time} and \texttt{cp\_dose}
        \item \textbf{Size:} 23,814 training rows: 21,948 treated and 1,866 controls (7.8\%)
        \item \textbf{Imbalance:} only 0.373\% of the (treated row, target) cells are positive
        \item \textbf{Metric:} mean multi-label binary log loss, clipped to avoid $\log(0)$
        \item Control rows (\texttt{cp\_type == ctl\_vehicle}) carry \textbf{no MoA}: excluded from training, set to 0 at submission
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Why an ensemble?}
        Blending different model families (dense net, attention, trees) fed with different preprocessing is a standard way to reduce variance. The blend gain is measured in this project; whether the models' errors really differ was \textbf{not} measured.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------
% PREPROCESSING
% ---------------------------------------------------------------
\begin{frame}{Two Preprocessing Pipelines}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{PCA variant (ResNet)}
        VarianceThreshold $\to$ RankGauss (\texttt{QuantileTransformer}) $\to$ PCA(50) on \texttt{g-}, PCA(15) on \texttt{c-} $\to$ \textbf{concatenate} transformed features, PCA components and raw dummies ($\approx$950 columns)
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Raw variant (FT-Transformer, CatBoost)}
        VarianceThreshold $\to$ RankGauss, \textbf{no PCA} -- keeps one token/split per original feature ($\approx$890 columns)
      \end{block}
    \end{column}
  \end{columns}
  \vspace{2mm}
  \small Both fitted \textbf{per fold} on the training split only and cached to disk as \texttt{.joblib}. The raw pipeline fitted for the transformer is reused by CatBoost within the same run; stale caches are cleared at the start of each run. The fitted pipelines are saved as artifacts, so the inference notebook only applies them (no refitting).
\end{frame}

    \begin{column}{0.42\textwidth}
      \begin{itemize}
        \item \textbf{Gradient boosting}: each tree corrects the residual error of the previous ones
        \item A single multi-output \texttt{MultiLogloss} fit, not 206 separate fits
        \item 250 symmetric trees of depth 5, learning rate 0.08
        \item Splits stay aligned to the original features $\to$ trained on \textbf{raw RankGauss} (no PCA)

\section{Training and Validation}

% ---------------------------------------------------------------
% METRIC, LOSS, TRAINING ROUTINE
% ---------------------------------------------------------------
\begin{frame}{Metric, Loss and Training Routine}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Evaluation metric}
      {\small
      \[
        \mathcal{L} = -\frac{1}{N \cdot 206}\sum_{i=1}^{N}\sum_{j=1}^{206}
        \Big[y_{ij}\log \hat y_{ij} + (1-y_{ij})\log(1-\hat y_{ij})\Big]
      \]}
      predictions clipped to $[\epsilon, 1-\epsilon]$, with $\epsilon = 10^{-15}$, computed in float64.
      \\
      \vspace{4mm}
      \textbf{Training loss}
      \begin{itemize}
        \item \texttt{BCEWithLogitsLoss} with \textbf{label smoothing} (0.001): $y(1-\epsilon) + 0.5\epsilon$
        \item Model selection \textbf{always} uses the plain, non-smoothed log loss
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item \textbf{AMP} (\texttt{torch.amp.GradScaler}) for speed and stability
        \item \textbf{Early stopping} (patience = 8 epochs) and \textbf{ReduceLROnPlateau}, both on the inner split
        \item Seeds set \emph{before} the model is built: initial weights depend on the seed
        \item \textbf{In-RAM checkpointing}: best \texttt{state\_dict} cloned to CPU, written to disk once at the end
        \item Explicit \texttt{torch.cuda.empty\_cache()} so sibling workers reclaim VRAM
      \end{itemize}
      \begin{alertblock}{Why it matters}
        Avoids dozens of slow disk writes per fold and prevents GPU memory creep across 30 jobs.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------
% VALIDATION PROTOCOL
% ---------------------------------------------------------------
\begin{frame}{Validation Protocol}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Outer loop: 5-fold OOF}
        \texttt{MultilabelStratifiedKFold}, seed 42, on treated rows. Each fold's validation part yields the out-of-fold predictions and is scored \textbf{once}, after training.
      \end{block}
      \begin{block}{Inner split: model selection}
        10\% of each training fold (stratified) drives early stopping, LR scheduling and best-epoch choice of the neural networks. CatBoost uses fixed iterations.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Blend: nested CV}
        Weights are fitted on $k-1$ folds' OOF predictions and scored on the held-out fold.
      \end{block}
      \begin{alertblock}{Known limitations}
        No drug identifier: the CV cannot be grouped by compound, so the same compound may appear in training and validation. Design choices were also made by looking at these same folds. Scores are optimistic (see the Kaggle test-set slide).
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------
% NN CV AT SCALE
% ---------------------------------------------------------------
\begin{frame}[fragile]{NN Cross-Validation at Scale}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item $2$ networks $\times$ $3$ seeds $\times$ $5$ folds $=$ \textbf{30 jobs}
        \item \texttt{joblib} + \texttt{loky} backend, GPUs assigned \textbf{round-robin}
        \item 2 GPUs $\times$ 2 jobs/GPU $=$ \textbf{4 parallel workers}
        \item Predictions averaged over the 3 seeds per (fold, model)
        \item Wall-clock: NN CV 98.5 min, CatBoost CV 29.1 min, inference 0.5 min (total 131 min)
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{lstlisting}
NN_JOBS_PER_GPU = 2
N_JOBS_NN = N_GPUS * NN_JOBS_PER_GPU

Parallel(n_jobs=N_JOBS_NN,
         backend="loky")(
  delayed(_train_nn_job)(...)
  for job in jobs
)
      \end{lstlisting}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Ensemble Blending: Method and Nested-CV Results}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \footnotesize
      \textbf{Weight search}
      \begin{itemize}
        \item Softmax-reparameterised weights on the simplex
        \item \textbf{Nelder--Mead}, not SLSQP: avoids a zero finite-difference gradient stalling the optimiser at $x_0$
      \end{itemize}
      \textbf{Nested CV}
      \begin{itemize}
        \item Weights fit on $k-1$ folds, scored on the held-out fold; final weights = mean across folds
        \item In-sample score (weights fit on all OOF rows): 0.01605, equal to the nested one; gap $9.4\cdot10^{-7}$. The blending step is not optimistic with three weights; this says nothing about leakage from the CV design

        \bottomrule
      \end{tabular}

      \vspace{2mm}
      \scriptsize CV scores use treated rows only. Control rows are predicted 0 (true 0) and add almost no loss, so the all-rows score $\approx$ treated-only score $\times$ treated share (0.910 in the public test set). Percentages are relative to the rescaled estimate.
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item \textbf{Late submission} through a separate \textbf{inference notebook}: loads 30 checkpoints, 5 CatBoost models, 10 fold pipelines and the blend weights; no training, no internet
        \item Predictions match the training notebook's (max.\ abs.\ difference $1.1\cdot10^{-16}$)
        \item Submitted \textbf{once}; the test set was never used for tuning
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{2mm}
  \begin{alertblock}{The CV estimate is optimistic}
    \small The private score is 11.8\% above the rescaled CV estimate. A likely cause is the missing grouping by compound (same compound in training and validation, different compounds in the test set) -- \textbf{not tested}. Public and private differ by 0.0023, over ten times the fold standard deviation: the fold std understates the uncertainty on unseen data.
  \end{alertblock}
\end{frame}

\begin{frame}{Conclusions, Limitations and Future Work}
  \small
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Findings}
      \begin{itemize}
        \item Ensemble 0.01605 vs.\ best single model (ResNet) 0.01632: $\approx$1.7\% better
        \item ResNet and FT-Transformer carry $\approx$82\% of the weight; CatBoost, weakest alone, gets 18\%
        \item An extra network (MLP or 1D-CNN) does not improve the blend
        \item Kaggle test set: private 0.01634, public 0.01864, against a rescaled CV estimate of 0.01461
      \end{itemize}
      \textbf{Future work}
      \begin{itemize}
        \item Group the CV by compound, or reserve a hold-out before tuning
        \item Ablations: PCA vs.\ raw for the ResNet, error correlation between models
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Limitations}
      \begin{itemize}
        \item No compound identifier: possible leakage across folds; the CV estimate is optimistic (confirmed by the Kaggle gap, cause not tested)
        \item Design choices made on the same folds that are reported; no hold-out reserved before tuning
        \item The inner split removes 10\% of the training data; an earlier protocol that selected epochs on the OOF fold gave 0.01595, a difference consistent with selection optimism but mixed with the data loss and noise
        \item GPU non-determinism: repeated runs differ by $\approx 10^{-5}$
        \item Why CatBoost earns weight (diverse errors) was not measured
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
